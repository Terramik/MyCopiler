

# скажем, что длинны всех входных векторов одинаковы


def v_print(v (float64[]) ) {
    var n uint64 = lenof v; var i uint64 = 0;
    print_s("[");
    while i < n {
        print_f("%9.3f ", v[i]);
        i = i + 1;
    }
    print_s("]\n");
}

def v_add(v1 (float64[]), v2 (float64[]), vr (float64[])) {
    var n uint64 = lenof v1; var i uint64 = 0;
    while i < n {
        vr[i] = v1[i] + v2[i];
        i = i + 1;
    }
}

def v_sub(v1 (float64[]), v2 (float64[]), vr (float64[])) {
    var n uint64 = lenof v1; var i uint64 = 0;
    while i < n {
        vr[i] = v1[i] - v2[i];
        i = i + 1;
    }
}

def v_mul(v1 (float64[]), v2 (float64[]), vr (float64[])) {
    var n uint64 = lenof v1; var i uint64 = 0;
    while i < n {
        vr[i] = v1[i] * v2[i];
        i = i + 1;
    }
}

def v_dot(v1 (float64[]), v2 (float64[])) -> (float64) {
    var n uint64 = lenof v1; var i uint64 = 0;
    var accumulator float64 = 0.0;
    while i < n {
        accumulator = accumulator + v1[i] * v2[i];
        i = i + 1;
    }
    return accumulator;
}

def sort(arr (float64[])) {
    var n uint64 = lenof arr;
    var i1 uint64 = 0;
    while i1 < n {
        var i2 uint64 = i1;
        var max_val float64 = arr[i2];
        var max_index uint64 = i2;

        while i2 < n {
            if arr[i2] > max_val {
                max_val = arr[i2];
                max_index = i2;
            }
            i2 = i2 + 1;
        }

        arr[i1], arr[max_index] = arr[max_index], arr[i1];

        i1 = i1 + 1;
    }
}

def main() -> (int64)
{
    var v1 (float64[4]) = [1.5, 2.1, 3.5, 4.1];
    var v2 (float64[4]) = [5.5, 6.1, 7.5, 8.1];
    var vr (float64[4]) = [1.5, 1.1, 2.5, 2.1];
    v_add(v1[:], v2[:], vr[:]);
    v_print(vr[:]);
    v_mul(v1[:], v1[:], vr[:]);
    v_print(vr[:]);
    print_f("%9.3f\n", v_dot(v1[:], v2[:]));
    var arr (float64[6]) = [3.14, 4.0, 5.0, 100.0, -300.0, 1.0*2*3];
    sort(arr[:]);
    v_print(arr[:]);
    return 0;
}
