

def nonconst(i int64) -> (int64) {
    return i * i * 3;
}

var glob_blob int64 = nonconst(3);

def main() {
    var i int64 = 10;
    def set_10() {
        i = 10;
    }
    def set_0() {
        i = 0;
    }
    print_i("start: %d\n", i);
    set_0();
    print_i("0: %d\n", i);
    set_10();
    print_i("10: %d\n", i);
    def sq(v int64) -> (int64) {
        return v * v;
    }
    print_i("sq: %d\n", sq(i) );
    print_i("glob_blob: %d\n", glob_blob );
}
